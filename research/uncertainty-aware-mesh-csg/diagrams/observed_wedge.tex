% Observed wedge: intended -> three quantization buckets -> two manifestations.

% Isometric projection from 3D (x,y,z) to 2D screen.
% +x and +y come toward the viewer; +z is up.
\newcommand{\iso}[3]{({((#1) - (#2)) * 0.866 * \sclI}, {((#3) - ((#1) + (#2)) * 0.5) * \sclI})}
